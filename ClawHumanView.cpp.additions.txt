#include "TouchOverlayEvent.h"
#include <memory>

// At the very beginning of ClawHumanView::VOnEvent, BEFORE HumanView::VOnEvent:
if (event.type == SDL_USEREVENT && event.user.code == OPENCLAW_TOUCH_OVERLAY_EVENT_CODE) {
    const int action = (int)(intptr_t)event.user.data1;
    OpenClawTouchPayload* raw = static_cast<OpenClawTouchPayload*>(event.user.data2);
    std::unique_ptr<OpenClawTouchPayload> payload(raw);
    if (payload && m_pActorController)
        m_pActorController->OnCustomOverlayAction(action, payload->pressed, payload->x, payload->y);
    return true;
}
